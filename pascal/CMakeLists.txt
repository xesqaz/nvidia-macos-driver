cmake_minimum_required(VERSION 3.16)
project(pascal_stage0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(PASCAL_SANITIZE "Enable address and undefined-behaviour sanitizers" OFF)
add_library(gp107 STATIC src/gp107.c)
target_include_directories(gp107 PUBLIC include)
add_executable(gp107-inspect tools/inspect.c)
target_link_libraries(gp107-inspect PRIVATE gp107)
add_executable(gp107-test tests/test_core.c)
target_link_libraries(gp107-test PRIVATE gp107)
foreach(target gp107 gp107-inspect gp107-test)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(PASCAL_SANITIZE)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
enable_testing()
add_test(NAME core COMMAND gp107-test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME tools COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tools.py)
endif()
